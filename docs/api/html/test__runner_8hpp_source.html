119 // Task submission
120 template <typename F, typename... Args>
121 auto submit(F&& f, Args&&... args) -> std::future<typename std::invoke_result<F, Args...>::type> {
122 using return_type = typename std::invoke_result<F, Args...>::type;
123
124 auto task = std::make_shared<std::packaged_task<return_type()>>(
125 std::bind(std::forward<F>(f), std::forward<Args>(args)...));
126
127 std::future<return_type> result = task->get_future();
128
129 {
130 std::unique_lock<std::mutex> lock(queue_mutex_);
131 if (stop_) {
132 throw std::runtime_error("Cannot submit task to stopped thread pool");
133 }
134
135 tasks_.emplace([task]() { (*task)(); });
136 }
137
138 condition_.notify_one();
139 return result;
140 }
141
142 // Pool management
143 void shutdown();
144 void wait_for_completion();
145 [[nodiscard]] size_t active_thread_count() const;
146 [[nodiscard]] size_t queued_task_count() const;
147
148 private:
149 std::vector<std::thread> workers_;
150 std::queue<std::function<void()>> tasks_;
151 std::mutex queue_mutex_;
152 std::condition_variable condition_;
153 std::atomic<bool> stop_;
154 std::atomic<size_t> active_tasks_;
