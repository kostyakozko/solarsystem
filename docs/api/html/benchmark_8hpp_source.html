209template <typename Func>
210BenchmarkResult Benchmark::measure(Func&& func, size_t iterations) {
211 if (iterations == 0) {
212 iterations = config_.default_iterations;
213 }
214
215 std::vector<std::chrono::nanoseconds> timings;
216 timings.reserve(iterations + config_.warmup_iterations);
217
218 size_t peak_memory = 0;
219
220 // Warmup iterations
221 for (size_t i = 0; i < config_.warmup_iterations; ++i) {
222 func();
223 }
224
225 // Actual benchmark iterations
226 for (size_t i = 0; i < iterations; ++i) {
227 size_t memory_before = 0;
228 if (config_.measure_memory) {
229 memory_before = get_current_memory_usage();
230 }
231
232 auto start = std::chrono::high_resolution_clock::now();
233 func();
234 auto end = std::chrono::high_resolution_clock::now();
235
236 if (config_.measure_memory) {
237 size_t memory_after = get_current_memory_usage();
238 peak_memory = std::max(peak_memory, memory_after - memory_before);
239 }
240
241 timings.push_back(std::chrono::duration_cast<std::chrono::nanoseconds>(end - start));
242 }
243
244 auto result = calculate_statistics(timings, peak_memory);
245 validate_thresholds(result);
246
247 return result;
248}
