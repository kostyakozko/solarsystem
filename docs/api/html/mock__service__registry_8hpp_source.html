467template <typename T>
468std::shared_ptr<T> ServiceScope::get_service() {
469 std::lock_guard<std::mutex> lock(scope_mutex_);
470
471 auto type_key = std::type_index(typeid(T));
472 auto it = scoped_instances_.find(type_key);
473
474 if (it != scoped_instances_.end()) {
475 return std::any_cast<std::shared_ptr<T>>(it->second);
476 }
477
478 // Get from registry
479 auto service = MockServiceRegistry::instance().resolve<T>();
480 if (service) {
481 scoped_instances_[type_key] = service;
482 }
483
484 return service;
485}

574template <typename T>
575std::shared_ptr<T> MockServiceRegistry::resolve(const std::string& name) {
576 std::lock_guard<std::mutex> lock(registry_mutex_);
577
578 auto key = make_service_key<T>(name);
579 auto it = services_.find(key);
580
581 if (it == services_.end()) {
582 return nullptr;
583 }
584
585 const auto& registration = it->second;
586
587 switch (registration.lifetime) {
588 case ServiceLifetime::Singleton:
589 return get_or_create_singleton<T>(registration);
590
591 case ServiceLifetime::Scoped:
592 if (current_scope_) {
593 return current_scope_->get_service<T>();
594 }
595 // Fall through to transient if no scope
596 [[fallthrough]];
597
598 case ServiceLifetime::Transient:
599 return create_service_instance<T>(registration);
600 }
601
602 return nullptr;
603}
